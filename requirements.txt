#
# SCOPE: everything the running FastAPI container needs, and nothing else.
# Offline/CI tooling lives in requirements-dev.txt — keep it OUT of here or
# pytest/ruff/mypy end up in the production image.
#
# Dev + offline-script deps that ARE pinned in requirements-dev.txt:
#   pytest, pytest-asyncio, pytest-mock, pytest-cov, ruff, mypy,
#   networkx, matplotlib   (scripts/visualize_graph.py + tests/unit/test_visualize_graph.py)

# ---------------------------------------------------------------------------
# !! torch variant pinning — READ BEFORE EDITING !!
# ---------------------------------------------------------------------------
# This stack is CPU-only by design (docs/09: Ollama CPU, bge-m3 on CPU, no GPU
# in the local/dev compose stack). But `torch==2.14.0` resolves to a DIFFERENT
# default per platform:
#
#   * Windows dev venv  -> torch 2.14.0+cpu   (506 MB, no nvidia_* packages)
#   * Linux slim (image)-> torch 2.14.0+cu13  (1.2 GB) + 16 nvidia_* packages
#                          = 3.2 GB of CUDA libraries that can NEVER be used,
#                            because there is no GPU and no CUDA runtime.
#
# Measured on this repo's image: site-packages 5.9 GB, of which nvidia_* =
# 3.2 GB. The image shipped at 6.24 GB. The extra 3.2 GB buys nothing.
#
# So the CPU wheel index is pinned explicitly below. `+cpu` is a LOCAL VERSION
# LABEL on a PyPI-published wheel — not a PEP 440 local version, and not
# `--find-links`. It resolves normally from PyPI with no extra index.
#
# Do NOT "simplify" this back to plain `torch==`. That silently reintroduces
# 3.2 GB of dead CUDA libraries on every Linux build, and nothing fails — the
# image just gets 5x bigger. This is pinned at requirements.txt:41-45 for the
# same reason docker-compose.yml pins the Ollama digest: bit-reproducibility.
# If a future GPU host needs CUDA, that is a deliberate new decision (separate
# requirements file / build arg), not a casual version edit here.
# ---------------------------------------------------------------------------

# API / Backend
fastapi==0.141.1       # POST /api/v1/ask + GET /api/v1/health; validasi Pydantic v2, X-Request-ID
uvicorn==0.54.0        # ASGI server; dipakai docker/backend.Dockerfile CMD
pydantic==2.13.5       # Skema AskRequest/AskResponse/EvidenceObject (docs/05 §4.1, docs/06 §5)

# Database — PostgreSQL / Supabase, role app_readonly
# Backend uses asyncpg exclusively (backend/app/core/db.py, 6 import sites).
# psycopg is NOT runtime: it is the sync driver for scripts/*.py + migrate.py,
# so it lives in requirements-dev.txt, not here.
asyncpg==0.31.0        # Pool async runtime FastAPI (statement_timeout 10s)

# RAG / Retrieval — SQL AST gate
sqlglot==30.20.0       # Validator whitelist 9 tabel kanonikal, root Select, LIMIT 50, COUNT(DISTINCT publication_id)

# Vector / Embedding — online query embedding (Phase 4, Task 6)
# Backend imports this lazily inside _load_sentence_transformer()
# (backend/app/services/embedding.py:78), NOT at module top level, so a missing
# wheel degrades to the Ollama bge-m3 HTTP fallback instead of crashing import.
# torch==2.14.0+cpu is pinned for the reason documented above: plain
# `torch==2.14.0` drags 3.2 GB of unusable CUDA libs into the Linux image.
torch==2.14.0+cpu
sentence-transformers==6.1.0  # Local BAAI/bge-m3 1024-dim query embedding; fallback endpoint embedding Ollama bila load lokal gagal

# Utilities
httpx==0.28.1           # Klien HTTP async untuk Ollama service & testing API
python-dotenv==1.2.3   # Baca DB_URL/OLLAMA_HOST untuk skrip lokal; di Compose nilai dari env_file (.env tak pernah di-commit)

# Observability — /metrics (Prometheus text format) untuk synthesis fallback counters.
# Dipisah dari stats.py yang sengaja stdlib-only, supaya logika counter tetap
# importable & testable tanpa dependency ini.
prometheus-client==0.26.0  # CollectorRegistry + generate_latest; custom collector bridging SynthesisStats
